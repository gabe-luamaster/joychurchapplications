const API_URL = "https://sheetdb.io/api/v1/0gowkb0zc5uc4"; const USERNAME = "jcistaff234"; const PASSCODE = "gabr123"; let applications = []; /* LOGIN / LOGOUT */ function login(){ const user = document.getElementById("username").value; const pass = document.getElementById("passcode").value; if(user === USERNAME && pass === PASSCODE){ document.getElementById("loginCard").classList.add("hidden"); document.getElementById("dashboard").classList.remove("hidden"); loadApplications(); } else { document.getElementById("loginError").classList.remove("hidden"); } } function logout(){ document.getElementById("loginCard").classList.remove("hidden"); document.getElementById("dashboard").classList.add("hidden"); } /* LOAD APPLICATIONS FROM SHEETDB */ async function loadApplications(){ try{ const res = await fetch(API_URL); applications = await res.json(); populateTeams(); renderTable(); }catch(err){ alert("Failed to load applications"); console.error(err); } } /* TEAM FILTER */ function populateTeams(){ const teamFilter = document.getElementById("teamFilter"); const teams = [...new Set(applications.map(a => a.team))]; teamFilter.innerHTML = '' + teams.map(t => ``).join(""); } /* RENDER TABLE */ function renderTable(){ const filter = document.getElementById("teamFilter").value; const tbody = document.querySelector("#appsTable tbody"); tbody.innerHTML = ""; applications.forEach((a,index)=>{ if(filter !== "all" && a.team !== filter) return; const tr = document.createElement("tr"); tr.innerHTML = ` ${a.team || ""} ${a.preferredName || ""} ${a.roblox || ""} ${a.discord || ""} ${a.scale || ""} ${a.qualified || ""} ${a.unavailable || ""} ${a.status || "Pending"} `; tbody.appendChild(tr); }); } /* APPROVE / DENY */ async function approve(index){ const robloxUser = applications[index].roblox; await fetch(`${API_URL}/roblox/${robloxUser}`,{ method:"PATCH", headers:{ "Content-Type":"application/json" }, body:JSON.stringify({ data:{status:"Approved"} }) }); loadApplications(); } async function deny(index){ const robloxUser = applications[index].roblox; await fetch(`${API_URL}/roblox/${robloxUser}`,{ method:"PATCH", headers:{ "Content-Type":"application/json" }, body:JSON.stringify({ data:{status:"Denied"} }) }); loadApplications(); } /* TEAM FILTER EVENT */ document.getElementById("teamFilter").addEventListener("change",renderTable);
