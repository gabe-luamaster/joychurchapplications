const PASSCODE = "gabr123"; // Your admin passcode let applications = []; // Will hold all merged applications /* LOGIN / LOGOUT */ function login(){ const input = document.getElementById("passcode").value; if(input === PASSCODE){ document.getElementById("loginCard").classList.add("hidden"); document.getElementById("dashboard").classList.remove("hidden"); loadApplications(); } else { document.getElementById("loginError").classList.remove("hidden"); } } function logout(){ document.getElementById("loginCard").classList.remove("hidden"); document.getElementById("dashboard").classList.add("hidden"); } /* MERGE LOCAL AND REMOTE APPLICATIONS */ async function loadApplications(){ try { // Fetch from nPoint const res = await fetch('https://api.npoint.io/8c134d701d74f33e8026'); const remoteApps = await res.json(); // Get localStorage apps const localApps = JSON.parse(localStorage.getItem("jciApps")||"[]"); // Merge: remote first, then local applications = [...remoteApps, ...localApps]; populateTeams(); renderTable(); } catch(err){ alert("Failed to load applications from nPoint."); console.error(err); // Fallback: use localStorage only applications = JSON.parse(localStorage.getItem("jciApps")||"[]"); populateTeams(); renderTable(); } } /* POPULATE TEAM FILTER */ function populateTeams(){ const teamFilter = document.getElementById("teamFilter"); const teams = [...new Set(applications.map(a=>a.team))]; teamFilter.innerHTML = '' + teams.map(t=>``).join(""); } /* RENDER TABLE */ function renderTable(){ const filter = document.getElementById("teamFilter").value; const tbody = document.querySelector("#appsTable tbody"); tbody.innerHTML = ""; applications.forEach((a,index)=>{ if(filter !== "all" && a.team !== filter) return; const tr = document.createElement("tr"); tr.innerHTML = ` ${a.team} ${a.preferredName} ${a.roblox} ${a.discord} ${a.scale} ${a.qualified} ${a.extra} TOS: ${a.agreements?.TOS || ''}
Comfort: ${a.agreements?.comfort || ''}
Training: ${a.agreements?.training || ''}
Admin: ${a.agreements?.adminAbuse || ''} ${a.status || 'Pending'} `; tbody.appendChild(tr); }); } /* APPROVE / DENY FUNCTIONS */ function approve(index){ applications[index].status = "Approved"; saveLocal(); renderTable(); } function deny(index){ applications[index].status = "Denied"; saveLocal(); renderTable(); } /* SAVE LOCAL MERGED DATA */ function saveLocal(){ // Only save local submissions back to localStorage const localApps = JSON.parse(localStorage.getItem("jciApps")||"[]"); const newLocal = applications.filter(a=>localApps.includes(a)); localStorage.setItem("jciApps", JSON.stringify(newLocal)); } /* TEAM FILTER CHANGE */ document.getElementById("teamFilter").addEventListener("change", renderTable);
